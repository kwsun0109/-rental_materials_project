2. src/pages/Materials.jsx
jsx
import { useState, useEffect } from "react";
import {
  Box, Typography, TextField, Button, Stack, Alert, IconButton,
} from "@mui/material";
import DeleteIcon from "@mui/icons-material/Delete";
import Table from "../components/Table";
import { getMaterials, createMaterial, deleteMaterial } from "../api";

const initialForm = { name: "", category: "", unit: "", total_qty: 0 };

function Materials() {
  const [materials, setMaterials] = useState([]);
  const [form, setForm] = useState(initialForm);
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");

  const load = () => {
    getMaterials().then((res) => setMaterials(res.data));
  };

  useEffect(() => { load(); }, []);

  const handleChange = (field) => (e) => {
    setForm({ ...form, [field]: e.target.value });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");
    setSuccess("");
    try {
      await createMaterial({
        ...form,
        total_qty: Number(form.total_qty) || 0,
      });
      setSuccess("자재가 등록되었습니다.");
      setForm(initialForm);
      load();
    } catch (err) {
      setError(err.message);
    }
  };

  const handleDelete = async (id) => {
    if (!window.confirm("정말 삭제하시겠습니까?")) return;
    try {
      await deleteMaterial(id);
      load();
    } catch (err) {
      setError(err.message);
    }
  };

  const columns = [
    { key: "id", label: "ID" },
    { key: "name", label: "자재명" },
    { key: "category", label: "분류" },
    { key: "unit", label: "단위" },
    { key: "total_qty", label: "총 보유 수량" },
    {
      key: "action",
      label: "삭제",
      render: (row) => (
        <IconButton size="small" color="error" onClick={() => handleDelete(row.id)}>
          <DeleteIcon fontSize="small" />
        </IconButton>
      ),
    },
  ];

  return (
    <Box>
      <Typography variant="h5" gutterBottom>자재 관리</Typography>

      {error && <Alert severity="error" sx={{ mb: 2 }}>{error}</Alert>}
      {success && <Alert severity="success" sx={{ mb: 2 }}>{success}</Alert>}

      <Box component="form" onSubmit={handleSubmit} sx={{ mb: 4, maxWidth: 480 }}>
        <Stack spacing={2}>
          <TextField
            label="자재명" value={form.name}
            onChange={handleChange("name")} required
          />
          <TextField
            label="분류" value={form.category}
            onChange={handleChange("category")}
            placeholder="예: 중장비, 공구 등"
          />
          <TextField
            label="단위" value={form.unit}
            onChange={handleChange("unit")}
            placeholder="예: 개, 대, kg"
          />
          <TextField
            label="초기 보유 수량" type="number" value={form.total_qty}
            onChange={handleChange("total_qty")}
          />
          <Button type="submit" variant="contained">등록</Button>
        </Stack>
      </Box>

      <Typography variant="h6" gutterBottom>등록된 자재 목록</Typography>
      <Table columns={columns} rows={materials} />
    </Box>
  );
}

export default Materials;

materials, companies를 맨 위로 올린 이유는 — 실제 사용 흐름상 자재와 거래처를 먼저 등록해야 반입/반출 등록 화면의 드롭다운(select)에 데이터가 나타나기 때문입니다.

이제 전체 화면 구성이 끝났습니다. 사용 순서는 이렇습니다:

자재 관리 → 자재 등록 (예: 굴삭기, 안전모)
거래처 관리 → 거래처 등록 (예: OO건설)
반입/반출 등록 → 위에서 등록한 자재/거래처로 실제 거래 기록
이후 재고 현황 / 반납 알림 / 정산 / 분실파손 화면에서 결과 확인

전체 배경을 보랏빛으로 바꿔드릴게요. MUI 테마 시스템을 활용해서 일관되게 적용하겠습니다.

1. src/theme.js (새 파일 생성)
javascript
import { createTheme } from "@mui/material/styles";

const theme = createTheme({
  palette: {
    background: {
      default: "#f3ecfa",  // 전체 배경 (연보라)
      paper: "#ffffff",    // 카드/테이블 등은 흰색 유지 (가독성)
    },
    primary: {
      main: "#7e57c2",     // 버튼, 강조 요소 (보라)
    },
  },
});

export default theme;

2. src/main.jsx 에 ThemeProvider 적용
jsx
import React from "react";
import ReactDOM from "react-dom/client";
import { ThemeProvider, CssBaseline } from "@mui/material";
import App from "./App.jsx";
import theme from "./theme";

ReactDOM.createRoot(document.getElementById("root")).render(
  <React.StrictMode>
    <ThemeProvider theme={theme}>
      <CssBaseline />
      <App />
    </ThemeProvider>
  </React.StrictMode>
);

CssBaseline이 theme.palette.background.default를 자동으로 <body>에 적용해줍니다.

추가 수정요령

파일별로 정확히 뭘 해야 하는지 표로 정리하면:

파일	작업	내용
src/theme.js	신규 생성	이 파일 자체가 없었으니 새로 만드는 파일입니다
src/main.jsx	기존 코드 수정	ThemeProvider, theme import 추가하고, 기존 렌더링 구조를 감싸는 형태로 변경
src/App.jsx	기존 코드 수정	기존에 있던 CssBaseline import와 사용 부분만 제거 (나머지 라우트 코드는 그대로 유지)
src/components/Sidebar.jsx	기존 코드 수정	Drawer의 sx 속성과 ListItemButton의 스타일 부분만 수정 (메뉴 항목, 라우팅 로직은 그대로)
요약하면
theme.js → 새 파일 만드세요 (src/theme.js 경로에)
main.jsx, App.jsx, Sidebar.jsx → 이미 갖고 계신 기존 파일을 열어서, 제가 드린 코드로 덮어쓰기(전체 교체) 하시면 됩니다. 
각 파일에서 바뀐 부분은 이렇습니다:

main.jsx: ThemeProvider로 감싸는 구조 추가
App.jsx: CssBaseline 제거 (한 줄)
Sidebar.jsx: 색상 스타일(sx) 부분만 추가

즉, 지금까지 만든 Materials.jsx, Companies.jsx, TransactionForm.jsx 등 다른 페이지 파일들은 전혀 손댈 필요 없습니다.
 저 3개 파일(theme.js 신규, main.jsx/App.jsx/Sidebar.jsx 수정)만 반영하시면 전체 화면 배경이 보라빛으로 바뀝니다.
