1. src/api/client.js
javascript
import axios from "axios";

const client = axios.create({
  baseURL: "http://localhost:8000",
  headers: {
    "Content-Type": "application/json",
  },
});

// 에러 응답 공통 처리
client.interceptors.response.use(
  (response) => response,
  (error) => {
    const message = error.response?.data?.detail || "서버 오류가 발생했습니다.";
    console.error("API Error:", message);
    return Promise.reject(new Error(message));
  }
);

export default client;

5. src/components/Table.jsx (재사용 테이블)
jsx
import {
  Table as MuiTable,
  TableBody,
  TableCell,
  TableContainer,
  TableHead,
  TableRow,
  Paper,
} from "@mui/material";

/**
 * columns: [{ key: 'name', label: '이름' }, { key: 'qty', label: '수량', render: (row) => ... }]
 * rows: 데이터 배열
 */
function Table({ columns, rows }) {
  return (
    <TableContainer component={Paper}>
      <MuiTable size="small">
        <TableHead>
          <TableRow>
            {columns.map((col) => (
              <TableCell key={col.key}>{col.label}</TableCell>
            ))}
          </TableRow>
        </TableHead>
        <TableBody>
          {rows.length === 0 ? (
            <TableRow>
              <TableCell colSpan={columns.length} align="center">
                데이터가 없습니다.
              </TableCell>
            </TableRow>
          ) : (
            rows.map((row, idx) => (
              <TableRow key={row.id ?? idx}>
                {columns.map((col) => (
                  <TableCell key={col.key}>
                    {col.render ? col.render(row) : row[col.key]}
                  </TableCell>
                ))}
              </TableRow>
            ))
          )}
        </TableBody>
      </MuiTable>
    </TableContainer>
  );
}

export default Table;
6. src/pages/TransactionForm.jsx (핵심 기능)
jsx
import { useState, useEffect } from "react";
import {
  Box, Typography, TextField, MenuItem, Button, Stack, Alert,
} from "@mui/material";
import { getMaterials, getCompanies, createTransaction } from "../api";

function TransactionForm() {
  const [materials, setMaterials] = useState([]);
  const [companies, setCompanies] = useState([]);
  const [form, setForm] = useState({
    material_id: "",
    company_id: "",
    type: "반입",
    qty: "",
    rental_start_date: "",
    rental_due_date: "",
    note: "",
  });
  const [error, setError] = useState("");
  const [success, setSuccess] = useState("");

  useEffect(() => {
    getMaterials().then((res) => setMaterials(res.data));
    getCompanies().then((res) => setCompanies(res.data));
  }, []);

  const handleChange = (field) => (e) => {
    setForm({ ...form, [field]: e.target.value });
  };

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError("");
    setSuccess("");
    try {
      await createTransaction({
        ...form,
        material_id: Number(form.material_id),
        company_id: Number(form.company_id),
        qty: Number(form.qty),
        rental_start_date: form.rental_start_date || null,
        rental_due_date: form.rental_due_date || null,
      });
      setSuccess("등록되었습니다.");
      setForm({ ...form, qty: "", note: "" });
    } catch (err) {
      setError(err.message);
    }
  };

  return (
    <Box maxWidth={480}>
      <Typography variant="h5" gutterBottom>
        반입/반출 등록
      </Typography>

      {error && <Alert severity="error" sx={{ mb: 2 }}>{error}</Alert>}
      {success && <Alert severity="success" sx={{ mb: 2 }}>{success}</Alert>}

      <form onSubmit={handleSubmit}>
        <Stack spacing={2}>
          <TextField
            select label="자재" value={form.material_id}
            onChange={handleChange("material_id")} required
          >
            {materials.map((m) => (
              <MenuItem key={m.id} value={m.id}>{m.name}</MenuItem>
            ))}
          </TextField>

          <TextField
            select label="거래처" value={form.company_id}
            onChange={handleChange("company_id")} required
          >
            {companies.map((c) => (
              <MenuItem key={c.id} value={c.id}>{c.name}</MenuItem>
            ))}
          </TextField>

          <TextField
            select label="구분" value={form.type}
            onChange={handleChange("type")} required
          >
            <MenuItem value="반입">반입</MenuItem>
            <MenuItem value="반출">반출</MenuItem>
          </TextField>

          <TextField
            label="수량" type="number" value={form.qty}
            onChange={handleChange("qty")} required
          />

          {form.type === "반출" && (
            <>
              <TextField
                label="임대 시작일" type="date" value={form.rental_start_date}
                onChange={handleChange("rental_start_date")}
                InputLabelProps={{ shrink: true }}
              />
              <TextField
                label="반납 예정일" type="date" value={form.rental_due_date}
                onChange={handleChange("rental_due_date")}
                InputLabelProps={{ shrink: true }}
              />
            </>
          )}

          <TextField
            label="비고" multiline rows={2} value={form.note}
            onChange={handleChange("note")}
          />

          <Button type="submit" variant="contained">등록</Button>
        </Stack>
      </form>
    </Box>
  );
}

export default TransactionForm;
7. src/pages/InventoryStatus.jsx
jsx
import { useState, useEffect } from "react";
import { Box, Typography } from "@mui/material";
import Table from "../components/Table";
import { getMaterials, getMaterialInventory } from "../api";

function InventoryStatus() {
  const [rows, setRows] = useState([]);

  useEffect(() => {
    const load = async () => {
      const { data: materials } = await getMaterials();
      const results = await Promise.all(
        materials.map((m) => getMaterialInventory(m.id).then((res) => res.data))
      );
      setRows(results);
    };
    load();
  }, []);

  const columns = [
    { key: "name", label: "자재명" },
    { key: "total_qty", label: "총 보유 수량" },
    { key: "available_qty", label: "가용 수량" },
  ];

  return (
    <Box>
      <Typography variant="h5" gutterBottom>재고 현황</Typography>
      <Table columns={columns} rows={rows} />
    </Box>
  );
}

export default InventoryStatus;

import { useState, useEffect } from "react";
import { Box, Typography, Button, Chip } from "@mui/material";

기본적으로 http://localhost:5173에서 실행됩니다. FastAPI(main.py)의 CORS 설정이 localhost:3000으로 되어 있는데, Vite는 기본 포트가 3000이 아니라 5173이니 백엔드 CORS 설정을 수정해야 합니다.

python
# main.py
app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:5173"],  # Vite 기본 포트로 변경
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

이제 전체 흐름(MySQL → FastAPI → React)이 다 연결됐습니다.
